const matchAns={L1:'R4',L2:'R6',L3:'R1',L4:'R2',L5:'R5',L6:'R8',L7:'R9',L8:'R7',L9:'R3',L10:'R10'};

function mLeft(id){
  document.querySelectorAll('[data-g="m1"]').forEach(el=>{
    if(!el.classList.contains('done')) el.classList.remove('sel');
  });
  M.sel=id;
  const el=document.querySelector(`[data-id="${id}"]`);
  if(el&&!el.classList.contains('done')) el.classList.add('sel');

function mRight(id){
  if(!M.sel) return;
  const left=document.querySelector(`[data-id="${M.sel}"]`);
  const right=document.querySelector(`[data-id="${id}"]`);
  if(!left||!right) return;
  M.pairs[M.sel]=id;
  left.classList.remove('sel'); left.classList.add('done');
  right.classList.add('done');
  M.sel=null;
}
function checkTask1(){
  let correct=0;
  for(let L in matchAns){
    const lEl=document.querySelector(`[data-id="${L}"]`);
    const rEl=document.querySelector(`[data-id="${M.pairs[L]}"]`);
    const correctR=document.querySelector(`[data-id="${matchAns[L]}"]`);
    if(M.pairs[L]===matchAns[L]){
      correct++;
      if(lEl) lEl.classList.add('correct-final');
      if(rEl) rEl.classList.add('correct-final');
    } else {
      if(lEl){lEl.classList.remove('done');lEl.classList.add('wrong-final');}
      if(rEl){rEl.classList.remove('done');rEl.classList.add('wrong-final');}
      if(correctR) correctR.style.outline='2px dashed var(--green)';
    }
  }
  const pts=correct*2;
  setScore(1,pts);
  fb('fb1',`${correct}/10 дұрыс сәйкестік. Балл: ${pts}/20`,correct===10?'ok':correct>=5?'part':'fail');

}

// ─── TASK 4: CALCULATIONS ───
// A: hex prism a=4, h=9 → P=24, Sbuy=216, Stab=3√3/2·16≈41.57, Stol≈299.1
// B: sq pyramid a=8, h=3 → l=√(9+16)=5, P=32, Sbuy=80, Stab=64, Stol=144
const stab_hex = 3*Math.sqrt(3)/2*16;
const task4Ans={
  'ca-1':{v:24,   tol:0.1},
  'ca-2':{v:216,  tol:1},
  'ca-3':{v:stab_hex, tol:1.5},
  'ca-4':{v:216+2*stab_hex, tol:3},
  'cb-1':{v:5,    tol:0.1},
  'cb-2':{v:32,   tol:0.1},
  'cb-3':{v:80,   tol:1},
  'cb-4':{v:64,   tol:0.1},
  'cb-5':{v:144,  tol:1},
};
function checkTask4(){
  let correct=0;
  const total=Object.keys(task4Ans).length;
  for(let id in task4Ans){
    const inp=document.getElementById(id);
    if(!inp) continue;
    const v=parseFloat(inp.value);
    const {v:exp,tol}=task4Ans[id];
    if(!isNaN(v)&&Math.abs(v-exp)<=tol){correct++;inp.className='ok';}
    else inp.className='fail';
  }
  const pts=Math.round((correct/total)*20);
  setScore(4,pts);
  fb('fb4',
    `${correct}/${total} қадам дұрыс. Балл: ${pts}/20\nА: P=24, S_бүй=216, S_таб≈${stab_hex.toFixed(1)}, S_тол≈${(216+2*stab_hex).toFixed(1)}\nБ: l=5, P=32, S_бүй=80, S_таб=64, S_тол=144`,
    correct>=total-1?'ok':correct>=Math.floor(total/2)?'part':'fail');
  showDesc('desc4');
}
function resetTask4(){

  document.getElementById('fb4').className='feedback';
  document.getElementById('desc4').className='descriptor';
}
